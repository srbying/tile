import { preview } from '/private/tmp/tile-issue-31/node_modules/vite/dist/node/index.js';
const started=performance.now();
const server=await preview({root:'/private/tmp/tile-issue-31',preview:{host:'127.0.0.1',port:4183,strictPort:true},logLevel:'error'});
process.on('message',async message=>{
 if(message==='stop') {await server.close();process.exit(0);}
 else process.send({id:message.id,cpu:process.cpuUsage(),memory:process.memoryUsage(),usage:process.resourceUsage(),uptime:process.uptime()});
});
process.send({ready:true,startupWallMs:performance.now()-started,cpu:process.cpuUsage()});
